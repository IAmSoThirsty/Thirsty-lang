318    def verify(
319        self, key: KeyFile | None = None, *, historical: bool = False
320    ) -> tuple[bool, str]:
321        """Verify integrity and, by default, reject superseded proof methods.
322
323        ``historical=True`` verifies only hash/signature authenticity. It does
324        not accept historical analysis as current semantic assurance, nor does
325        either mode independently replay or prove the recorded statements.
326        """
327        expected_hash = self.compute_hash()
328        if self.record_hash != expected_hash:
329            return False, "record hash mismatch"
330        if not historical:
331            if self.engine_version != ENGINE_VERSION:
332                return False, "superseded admission engine; reanalysis required"
333            if any(
334                item.status == EvidenceStatus.PROVEN
335                and (
336                    item.analyzer != "CanonicalConvergence"
337                    or item.method not in _CURRENT_PROOF_METHODS
338                    or item.details.get("assumptions")
339                )
340                for item in self.evidence
341            ):
342                return (
343                    False,
344                    "unsupported proof method or assumptions; reanalysis required",
345                )
346        if not self.signature:
347            if self.signature_alg or self.signer_key_id:
348                return False, "incomplete signature metadata"
349            return True, (
350                "historical hash is valid; semantic assurance was not verified"
351                if historical
352                else "unsigned record hash is valid"
353            )
354        if self.signature_alg != "ed25519":
355            return False, "unsupported signature algorithm"
356        if key is None:
357            return False, "public key is required for signed record"
358        if key.role != ROLE_PROOF_SIGNER or key.key_id != self.signer_key_id:
359            return False, "signer key identity or role mismatch"
360        try:
361            signature = bytes.fromhex(self.signature)
362        except ValueError:
363            return False, "signature is not canonical hexadecimal"
364        if self.signature != self.signature.lower():
365            return False, "signature is not canonical lowercase hexadecimal"
366        try:
367            key.public_key().verify(signature, bytes.fromhex(self.record_hash))
368        except (InvalidSignature, ValueError):
369            return False, "invalid Ed25519 signature"
370        return True, (
371            "historical signature is valid; semantic assurance was not verified"
372            if historical
373            else "signed record is valid"
374        )

233    @classmethod
234    def parse(cls, text: str) -> ShadowModule:
235        """Parse mutation source text into a ShadowModule."""
236        if len(text) > _MAX_SOURCE_CHARS:
237            raise ValueError("mutation exceeds the analysis source-size limit")
238        text = text.strip()
239        mut_match = cls.MUTATION_RE.search(text)
240        if not mut_match:
241            raise ValueError("No valid mutation definition found")
242
243        name = mut_match.group(1)
244        validated_canonical_body = mut_match.group(2)
245
246        module = ShadowModule(name=name, source=text)
247        shadow_code = ""
248        invariant_code = ""
249        canonical_code = ""
250
251        for block_match in cls.BLOCK_RE.finditer(validated_canonical_body):
252            block_type = block_match.group(1).strip()
253            block_content = block_match.group(2).strip()
254
255            if block_type == "shadow":

257                module.shadow_code = shadow_code
258            elif block_type == "invariant":

260                module.invariant_code = invariant_code
261            elif block_type == "canonical":

263                module.canonical_code = canonical_code
264

266        if not shadow_code and not invariant_code and not canonical_code:
267            direct_blocks = cls.BLOCK_RE.finditer(text[len(f"mutation {name}") :])
268            for block_match in direct_blocks:
269                block_type = block_match.group(1).strip()
270                block_content = block_match.group(2).strip()
271                if block_type == "shadow":
272                    module.shadow_code = block_content
273                elif block_type == "invariant":
274                    module.invariant_code = block_content
275                elif block_type == "canonical":
276                    module.canonical_code = block_content
277
278        # Prefer the language's own grammar: parse the whole mutation through
279        # the real parser and lift the genuine shadow/invariant/canonical blocks
280        # off the ShadowThirstMutation node. Falls back to the per-block parse
281        # already done in ShadowModule.__post_init__ when this doesn't apply.

287                lexer.errors

294
